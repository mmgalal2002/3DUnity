public static class ProShieldBridge {
 public static Room ToRoom(Design d) {
  var r=new Room{width=d.width,height=d.depth,roomHeight=d.height,sampleDistance=d.sampleDistance}; r.floor=new HorizontalBarrier{thickness=d.floor.thickness,material=Material(d.floor.material)}; r.ceiling=new HorizontalBarrier{thickness=d.ceiling.thickness,material=Material(d.ceiling.material)};
  foreach(var i in d.items) {
   if(i==null)continue;
   if(i.kind=="Wall") { var a=i.angle*Math.PI/180.0; var dx=Math.Cos(a)*i.length/2.0; var dz=Math.Sin(a)*i.length/2.0; r.walls.Add(new Wall{id=i.id,x1=i.x-dx,y1=i.z-dz,x2=i.x+dx,y2=i.z+dz,thickness=i.shielding.thickness,material=Material(i.shielding.material)}); }
   else if(i.kind=="LINAC") { var e=ProShieldCatalog.DefaultEquipment(EquipmentType.linac,i.id,i.name); e.x=i.x;e.y=i.z;e.z=i.y+d.isocentreHeight;e.planRotationDegrees=i.angle;e.gantryAngleDegrees=d.gantry;e.workload=d.workload;e.kVp=d.energy*1000;e.linacFieldSizeCm=(d.fieldX+d.fieldY)/2.0;r.equipment.Add(e); }
   else if(i.kind=="Desk") r.workstations.Add(new Workstation{id=i.id,name=i.name,x=i.x,y=i.z,z=i.y+i.assessmentHeight,occupancyFactor=i.occupancy,isControlled=i.occupancy>=.5});
  }
  if(d.regions!=null)foreach(var q in d.regions){var region=new OccupiedRegion{id=q.id,name=q.name,occupancyFactor=q.occupancy,designGoal=q.designGoal,scope=ParseScope(q.scope)};foreach(var p in q.points)region.points.Add(new Point2D(p.x,p.y));r.occupiedRegions.Add(region);}
  return r;
 }
 public static ProjectEnvelope Envelope(Design d){return new ProjectEnvelope{savedAtUtc=DateTime.UtcNow.ToString("o"),room=ToRoom(d),editorDesign=d};}
 static WallMaterial Material(string s){switch((s??"").ToLowerInvariant()){case "lead":return WallMaterial.lead;case "gypsum":return WallMaterial.gypsum;case "steel":return WallMaterial.steel;case "glass":return WallMaterial.glass;case "wood":return WallMaterial.wood;default:return WallMaterial.concrete;}}
 static RegionScope ParseScope(string s){switch((s??"").ToLowerInvariant()){case "floor":return RegionScope.floor;case "ceiling":return RegionScope.ceiling;default:return RegionScope.wall;}}
}
